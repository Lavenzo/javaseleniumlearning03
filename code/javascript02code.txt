    let month = this.genNum(1, 12);
    let day;
    if (month === 4 || month === 6 || month === 9 || month === 11) {
      day = this.genNum(1, 30);
    } else if (month === 2) {
      day = this.genNum(1, 28);
    } else {
      day = this.genNum(1, 31);
    }
    let year = new Date().getFullYear();
    year = year - Number(age);
    month = String(month).padStart(2, "0");
    day = String(day).padStart(2, "0");
    let dateOfBirth = `${year}-${month}-${day}`;
    return dateOfBirth;

    await test.step(`Step 3: User clicks on the "${btn}" button`, async () => {
      await this.page.getByTestId("btn-click").click();
    });

    let timeStampChecker = await this.page
      .getByTestId("output-display")
      .textContent();
    timeStampChecker = timeStampChecker.trim();

      await expect(timeStampChecker).toContain(x);

    let clickTime = new Date();
    let clickDate = clickTime.toDateString();

    let clickTimestamp = new Date(`${clickDate} ${timeStampChecker}`);
    let currentTimestamp = new Date();

    let difference = Math.abs((currentTimestamp - clickTimestamp) / 1000);
    await test.step(`Step 6: Verify the Expected Result and Timestamp captured to be within 5 seconds`, async () => {
      expect(difference).toBeLessThanOrEqual(5);
    });
