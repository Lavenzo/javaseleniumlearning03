const { test, expect } = require("@playwright/test");

class Project03_HelperClass_pt01 {
  constructor(page) {
    this.page = page;
  }
  async navigateWebsiteScrollToSection(sectionLocator) {
    await test.step("Step 1: Navigate to CURA Healthcare Service website", async () => {
      await this.page.goto("https://katalon-demo-cura.herokuapp.com/");
    });

    await test.step("Step 2: Scroll to section", async () => {
      await sectionLocator.scrollIntoViewIfNeeded();
    });
  }
  async verifyHeaderLocator(headerLocation, headerTextResult) {
    await test.step(`Step 3: Verify page loaded successfully with correct ${headerLocation} heading`, async () => {
      const heading = this.page.locator(headerLocation);
      await expect(heading).toHaveText(headerTextResult);
    });
  }
  async verifyLiLocator(liLocator, expectedResult, liTestStep) {
    await test.step(`Step 3 Validate the li contents - ${liTestStep}`, async () => {
      await expect(liLocator).toContainText(expectedResult);
    });
  }
  async verifyIconVisibility(iconLocator, iconName) {
    await test.step(`Step 3: Verify ${iconName} icon is displayed`, async () => {
      await expect(iconLocator).toBeVisible;
    });
  }
  async clickHamburgerMenu() {
    await test.step("Step 1: Navigate to CURA Healthcare Service website", async () => {
      await this.page.goto("https://katalon-demo-cura.herokuapp.com/");
    });

    await test.step("Step 2: Click on side menu (hamburger icon)", async () => {
      const menuIcon = this.page.locator("i.fa.fa-bars");
      await menuIcon.click();
    });
  }
  async verifySidePanelHyperlinks(locatorAttribute, locatorText) {
    await test.step("Step 3: Verify hyperlink locator'", async () => {
      const navLink = this.page.locator("a", { hasText: locatorText });

      await expect(navLink).toHaveAttribute("href", locatorAttribute);
    });

    await test.step(`Step 4: Verify link text is "${locatorText}"`, async () => {
      const navLink = this.page.locator("a", { hasText: locatorText });

      await expect(navLink).toHaveText(locatorText);
    });
  }
  async checkUserNavigationLogin(headingType, headingWord) {
    await test.step("Step 4: Verify Login page header", async () => {
      const loginHeader = this.page.locator(headingType);
      await expect(loginHeader).toHaveText(headingWord);
    });
  }
}

const { test, expect } = require("@playwright/test");

class Project03_HelperClass_pt03 {
  constructor(page) {
    this.page = page;
  }
  async loginCURAWebsite() {
    await test.step("Step 1: Navigate to CURA Healthcare Service login page", async () => {
      await this.page.goto("https://katalon-demo-cura.herokuapp.com/");
      const btnLocator = this.page.locator("#btn-make-appointment");
      await btnLocator.click();
    });
  }
  async verifyHeaderLocator(headerLocation, headerTextResult) {
    await test.step(`Step 2: Verify user login successfully with correct ${headerLocation} heading`, async () => {
      const heading = this.page.locator(headerLocation);
      await expect(heading).toHaveText(headerTextResult);
    });
  }
  async fillUpLoginCredentials() {
    await test.step("Step 3: User fill up user name + password and click the login button", async () => {
      await this.page.locator("#txt-username").fill("John Doe");
      await this.page.locator("#txt-password").fill("ThisIsNotAPassword");
      await this.page.locator("#btn-login").click();
    });

    await test.step("Step 4: Verify user login successfully", async () => {
      const successHeader = this.page.locator("h2");
      await expect(successHeader).toHaveText("Make Appointment");
    });
  }
  async verifyAppointmentBooking(
    comboFacilityValue,
    hospitalReadmission,
    healthcareSelection,
    formattedDate,
    appointmentComment,
  ) {
    await test.step("Step 8: Validate appointment confirmation", async () => {
      const confirmationHeader = this.page.locator("h2");
      const infoText = this.page.locator("p.lead");

      await expect(confirmationHeader).toHaveText("Appointment Confirmation");
      await expect(infoText).toContainText("appointment has been booked");
    });

    await test.step("Step 9: Verify appointment confirmation details", async () => {
      await expect(this.page.locator("p.lead")).toHaveText(
        "Please be informed that your appointment has been booked as following:",
      );

      await expect(this.page.locator("#facility")).toHaveText(
        comboFacilityValue,
      );
      await expect(this.page.locator("#hospital_readmission")).toHaveText(
        hospitalReadmission,
      );
      await expect(this.page.locator("#program")).toHaveText(
        healthcareSelection,
      );
      await expect(this.page.locator("#visit_date")).toHaveText(formattedDate);
      await expect(this.page.locator("#comment")).toHaveText(
        appointmentComment,
      );
    });
  }
  computeRandDate() {
    const start = new Date(2025, 0, 1);
    const end = new Date(2026, 4, 31);
    const randomDate = new Date(
      start.getTime() + Math.random() * (end.getTime() - start.getTime()),
    );

    const day = String(randomDate.getDate()).padStart(2, "0");
    const month = String(randomDate.getMonth() + 1).padStart(2, "0");
    const year = randomDate.getFullYear();
    let formattedDate = `${day}/${month}/${year}`;
    return formattedDate;
  }
  async fillUpApplicationForm(
    comboFacilityValue,
    healthcareSelection,
    hospitalReadmission,
    formattedDate,
    appointmentComment,
  ) {
    await test.step("Step 5: Select facility", async () => {
      await this.page
        .locator("#combo_facility")
        .selectOption(comboFacilityValue);
    });
    await test.step("Step 6: Validate checkbox, radio, and set date", async () => {
      const readmissionCheckbox = this.page.locator(
        "#chk_hospotal_readmission",
      );

      const radioMap = {
        Medicare: "#radio_program_medicare",
        Medicaid: "#radio_program_medicaid",
        None: "#radio_program_none",
      };

      const healthcareRadio = this.page.locator(radioMap[healthcareSelection]);

      await healthcareRadio.check();
      await expect(healthcareRadio).toBeChecked();

      const dateInput = this.page.locator("#txt_visit_date");

      if (hospitalReadmission === "Yes") {
        if (!(await readmissionCheckbox.isChecked())) {
          await readmissionCheckbox.check();
        }
      } else {
        if (await readmissionCheckbox.isChecked()) {
          await readmissionCheckbox.uncheck();
        }
      }

      await expect(readmissionCheckbox).toHaveJSProperty(
        "checked",
        hospitalReadmission === "Yes",
      );

      await dateInput.click();
      await dateInput.fill("");
      await dateInput.type(formattedDate);
      await expect(dateInput).toHaveValue(formattedDate);
    });
    await test.step("Step 7: Enter comment and submit", async () => {
      await this.page.locator("#txt_comment").fill(appointmentComment);
      await this.page.locator("#btn-book-appointment").click();
    });
  }
}
