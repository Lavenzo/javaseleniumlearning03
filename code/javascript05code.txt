import { test, expect } from "@playwright/test";

export class project05_HelperClass_pt01 {
  constructor(page) {
    this.page = page;
  }
  async navigateWebPage(loginHeader) {
    await test.step("Step 1: Navigate to Swag Labs website", async () => {
      await this.page.goto("https://www.saucedemo.com/");
    });

    await test.step("Step 2: Verify user has navigated to the website", async () => {
      const loginHeader = this.page.locator(".login_logo");
      await expect(loginHeader).toHaveText("Swag Labs");
    });
  }
  async tc002_fillUpWrongPassword() {
    await test.step("Step 3: User enters invalid credentials and click login", async () => {
      await this.page.fill("#user-name", "standard_user");
      await this.page.fill("#password", "wrongpassword");
      await this.page.click("#login-button");
    });

    await test.step("Step 4: Verify error message", async () => {
      const errorMsg = this.page.locator('[data-test="error"]');
      await expect(errorMsg).toHaveText(
        "Epic sadface: Username and password do not match any user in this service",
      );
    });
  }
  async fillUpCorrectPassword(userName, userPwd) {
    await test.step(`Step 3: User enters userID: "${userName}" with correct password and click login`, async () => {
      await this.page.fill("#user-name", userName);
      await this.page.fill("#password", userPwd);
      await this.page.click("#login-button");
    });

    await test.step("Step 4: Verify user has login successfully", async () => {
      if (userName === "locked_out_user") {
        const errorMsg = this.page.locator('[data-test="error"]');
        await expect(errorMsg).toHaveText(
          "Epic sadface: Sorry, this user has been locked out.",
        );
      } else {
        const productsTitle = this.page.locator('[data-test="title"]');
        await expect(productsTitle).toHaveText("Products");
      }
    });
  }
}

}) => {
  // User navigate to the Swags lab website
  const logoLocator = page.locator(".login_logo");
  await prj05_pt02.navigateWebPage(logoLocator);

  // User fills up the ID and password
  const userName = "standard_user";
  const userPwd = "secret_sauce";
  await prj05_pt02.fillUpCorrectPassword(userName, userPwd);

  // Verify web navigate via the Twitter Link icon

  await test.step("Step 5: Click Twitter link icon and switch to new tab", async () => {
    const twitterIcon = page.locator('[data-test="social-twitter"]');
    await twitterIcon.scrollIntoViewIfNeeded();
    const [newPage] = await Promise.all([
      page.context().waitForEvent("page"),
      twitterIcon.click(),
    ]);

    await newPage.waitForLoadState();
    page = newPage;
  });

  await test.step("Step 6: Verify navigation to Twitter page", async () => {
    await expect(page).toHaveURL(/twitter\.com|x\.com/);

    await expect(
      page.getByRole("link", { name: "Sauce Labs" }).first(),
    ).toBeVisible();
  });
});
