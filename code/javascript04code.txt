export class project04_HelperClass_pt01 {
  constructor(page) {
    this.page = page;
  }
  async navigateWebPage(loginHeader) {
    await test.step("Step 1: Navigate to Webpage", async () => {
      await this.page.goto(
        "https://opensource-demo.orangehrmlive.com/web/index.php/auth/login",

    });

    await test.step("Step 2: Verify User navigated successfully to Orange HRM Webpage", async () => {
      await expect(loginHeader).toBeVisible();
      await expect(loginHeader).toHaveText("Login");
    });

  async tc002_FillIncorrectCredentials(loginID, loginPwd, expectedErrorMsg) {
    const usernameInputLocator = this.page.locator('input[name="username"]');
    const passwordInputLocator = this.page.locator('input[name="password"]');
    const loginButtonLocator = this.page.locator('button[type="submit"]');
    const errorMessageLocator = this.page.locator("p.oxd-alert-content-text");

    await test.step("Step 3: Enter incorrect credentials and click Login", async () => {
      await usernameInputLocator.fill(loginID);
      await passwordInputLocator.fill(loginPwd);
      await loginButtonLocator.click();
    });

    await test.step("Step 4: Verify invalid credentials message", async () => {
      await expect(errorMessageLocator).toBeVisible();
      await expect(errorMessageLocator).toHaveText(expectedErrorMsg);
    });
  }
  async tc003_FillCorrectCredentials(loginID, loginPwd) {
    const usernameInputLocator = this.page.locator('input[name="username"]');
    const passwordInputLocator = this.page.locator('input[name="password"]');
    const loginButtonLocator = this.page.locator('button[type="submit"]');
    const userDropdownNameLocator = this.page.locator(
      "p.oxd-userdropdown-name",
    );

    await test.step("Step 3: Enter correct credentials and click Login", async () => {
      await usernameInputLocator.fill(loginID);
      await passwordInputLocator.fill(loginPwd);
      await loginButtonLocator.click();

    await test.step("Step 4: Verify successful login and user name", async () => {

  constructor(page) {

  async fillCorrectCredentials(loginID, loginPwd) {

    const passwordInputLocator = this.page.locator('input[name="password"]');
    const loginButtonLocator = this.page.locator('button[type="submit"]');
    const userDropdownNameLocator = this.page.locator(
      "p.oxd-userdropdown-name",
    );

    await test.step("Step 3: Enter correct credentials and click Login", async () => {

      await expect(userDropdownNameLocator).toBeVisible();
      await this.page.waitForTimeout(3000);
    });
  }
  async tc009_navigateTimeSheetPage() {
    await test.step("Step 5: Click on stopwatch icon to access the attendance page", async () => {
      await this.page.locator("button:has(i.bi-stopwatch)").click();
    });

    await test.step("Step 6: Check user navigation to the attenance page", async () => {
      const punchInHeaderLocator = this.page.locator("h6.orangehrm-main-title");
      await expect(punchInHeaderLocator).toHaveText("Punch In");
      await this.page.waitForTimeout(1000);
    });
  }
  getRandomDate() {
    const start = new Date(2025, 0, 1);
    const end = new Date(2026, 4, 31);
    const randomDate = new Date(
      start.getTime() + Math.random() * (end.getTime() - start.getTime()),
    );
    const year = randomDate.getFullYear();
    const month = String(randomDate.getMonth() + 1).padStart(2, "0");
    const day = String(randomDate.getDate()).padStart(2, "0");
    const formattedDate = `${year}-${month}-${day}`;
    return formattedDate;
  }
  getRandomTime() {
    const hour = this.getHour();
    const minute = this.getMin();
    const formattedTime = `${String(hour).padStart(2, "0")}:${minute}`;
    return formattedTime;
  }
  getHour() {
    const hour = Math.floor(Math.random() * 24);
    return hour;
  }
  getMin() {
    const minute = String(Math.floor(Math.random() * 60)).padStart(2, "0");
    return minute;
  }
  async tc010_fillUpPunchInTimesheet(
    formattedDate,
    formattedTime,
    hour,
    minute,
  ) {
    // Fill date - robust placeholder handling
    const dateInputLocator = this.page
      .locator('input[placeholder*="yyyy"]')
      .or(this.page.locator('input[placeholder*="mm-dd-yyyy"]'))
      .or(this.page.locator('input[placeholder*="yyyy-dd-mm"]'))
      .first();

    await dateInputLocator.waitFor({ state: "visible" });
    await dateInputLocator.click();
    await dateInputLocator.fill(formattedDate);
    await dateInputLocator.dispatchEvent("change");
    await this.page.waitForTimeout(300);

    // Fill time using time picker (proven working method)
    const timeInputLocator = this.page
      .locator('input[placeholder="hh:mm"]')
      .first();

    console.log("Filling time using time picker...");
    await timeInputLocator.click();
    await this.page.waitForTimeout(500);

    // Select hour from time picker

    const hourOptions = this.page
      .locator('.oxd-time-picker-hour-option, [role="option"]')
      .filter({ hasText: String(hour).padStart(2, "0") });

    if ((await hourOptions.count()) > 0) {
      await hourOptions.first().click();
      await this.page.waitForTimeout(300);
    }

    // Select minute from time picker
    const minuteOptions = this.page
      .locator('.oxd-time-picker-minute-option, [role="option"]')
      .filter({ hasText: String(minute).padStart(2, "0") });

    if ((await minuteOptions.count()) > 0) {
      await minuteOptions.first().click();
      await this.page.waitForTimeout(300);
    }

    // Confirm time selection
    const confirmBtn = this.page.locator(
      'button:has-text("OK"), button:has-text("Confirm"), .oxd-button--success',
    );
    if ((await confirmBtn.count()) > 0) {
      await confirmBtn.first().click();
      await this.page.waitForTimeout(500);
    }

    // Verify time was filled
    const actualValue = await timeInputLocator.inputValue();
    console.log("Time Field Value After Selection:", actualValue);

    // Fill note
    const noteTextAreaLocator = this.page.locator(
      'textarea[placeholder="Type here"]',
    );
    const noteText = `Punch In Attendance Record\nPunch In Date: ${formattedDate}\nPunch In Time: ${formattedTime}`;
    await noteTextAreaLocator.fill(noteText);

    console.log("Generated Date:", formattedDate);
    console.log("Generated Time:", formattedTime);

    // Verification - all fields are filled
    await test.step("Step 8: Verify all input fields are filled", async () => {
      const dateInput = this.page
        .locator('input[placeholder*="yyyy"]')
        .or(this.page.locator('input[placeholder*="mm-dd-yyyy"]'))
        .or(this.page.locator('input[placeholder*="yyyy-dd-mm"]'))
        .first();

      const timeInput = this.page.locator('input[placeholder="hh:mm"]').first();
      const noteTextArea = this.page.locator(
        'textarea[placeholder="Type here"]',
      );

      await this.page.waitForTimeout(1000);

      await expect(dateInput).not.toHaveValue("");
      await expect(timeInput).not.toHaveValue("");
      await expect(noteTextArea).not.toHaveValue("");
    });
  }
}
